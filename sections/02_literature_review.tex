\subsection[EN e CH]{Trabalhos sobre \textit{corpora} conversacionais em Inglês e Chinês}

\begin{frame}{DailyTalk -- ICASSP 2023}
  \begin{block}{Contribuições}
    \begin{enumerate}
      \item \textit{Córpus} conversacional aberto para treinamento de modelos de síntese
        de fala conversacional;
      \item Proposta de modelo de síntese de fala conversacional sensível ao contexto;
      \item Métricas de avaliação de diálogos sintetizados por modelos de \textit{TTS}.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{DailyTalk -- ICASSP 2023}
  \begin{block}{}
    \begin{enumerate}
      \item<1-> Obtenção e Estruturação do \textit{córpus}.
      \item<2-> Modelo \textit{baseline} (\textbf{FastSpeech2} \cite{FastSpeech2} Modificado)
      \item<3-> Experimentos: integridade e adaptabilidade.
      \item<4-> Métricas:
        \begin{itemize}
          \item \textit{Dialogue-level MOS}
          \item \textit{Pairwise MOS} (GT-synthesized)
          \item \textit{Pairwise MOS} (synthesized pair)
          \item \textit{User-system Simulation MOS}
        \end{itemize}
    \end{enumerate}
  \end{block}
\end{frame}
